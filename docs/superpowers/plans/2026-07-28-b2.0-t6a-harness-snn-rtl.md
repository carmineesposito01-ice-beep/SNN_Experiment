**Files:**

- [ ] **Step 1: scrivere `tier_block_params.m`**

Pilota il **blocco reale** `Donatello_Tier@BALANCED` in Simulink (pattern `drive` di `run_block_traj_test`),
misura la latenza sull'ingresso costante (edge-trigger → 1 sola inferenza), campiona i 5 param a fine control-step.

```matlab
function [P, lat] = tier_block_params(trajList, hold, nfrac)
%TIER_BLOCK_PARAMS  Oracolo T6: gira Donatello_Tier@BALANCED (nfrac Base 13) in Simulink su ogni
%  traiettoria di test_dataset.mat, campiona i 5 param [v0 T s0 a b] a fine control-step. Fedele al
%  blocco PER COSTRUZIONE (e' il blocco). Ritorna P{i} = N_i x 5 (double) e la latenza misurata.
%  hold  : clock/control-step tenuti (>= latenza ~364 del BALANCED splitpipe); default 500 (== TB RTL).
%  nfrac : bit frazionari ingresso (default 20 = bit-exact; <20 e' il controllo negativo di HDL_PHASE §3.1.3).
  if nargin<2 || isempty(hold),  hold  = 500; end
  if nargin<3 || isempty(nfrac), nfrac = 20;  end
  here = fileparts(mfilename('fullpath'));
  mroot = fullfile(here, '..', '..', 'matlab'); addpath(mroot);
  ds = load(fullfile(mroot,'test_dataset.mat')); tr = ds.trajectories;
  % latenza: ingresso costante -> UNA sola inferenza (edge-trigger). Misurata, non assunta.
  v1 = double(tr{trajList(1)}.val); v1 = v1(:,1);
  Plog = drive_tier(v1*ones(1,3), 700, 700, nfrac);
  chg  = find(any(abs(diff(Plog,1,1))>0, 2));
  assert(~isempty(chg), 'il blocco non produce output (FSM ferma?)');
  assert(numel(chg)==1, 'ingresso costante -> %d inferenze: non e'' edge-triggered', numel(chg));
  lat = chg(1);
  assert(hold >= lat, 'hold=%d < latenza=%d', hold, lat);
  fprintf('tier_block_params: latenza BAL misurata = %d clock (edge-trigger OK)\n', lat);
  P = cell(numel(trajList),1);
  for i = 1:numel(trajList)
    val = double(tr{trajList(i)}.val); K = size(val,2);
    Pall = drive_tier(val, hold, K*hold+20, nfrac);
    idx  = hold*(0:K-1).' + lat + 1; idx = idx(idx <= size(Pall,1));
    assert(numel(idx)==K, 'traj %d: campionati %d/%d control-step', trajList(i), numel(idx), K);
    P{i} = Pall(idx,:);
  end
end

function P = drive_tier(seq, hold, stopT, nfrac)
% pilota Donatello_Tier@BALANCED@13 con la sequenza fisica seq (4 x K), ogni colonna tenuta `hold` clock.
% (specializzazione del pattern drive di run_block_traj_test: forza esplicitamente la variante).
  if nargin<3||isempty(stopT), stopT = hold; end
  K = size(seq,2);
  assignin('base','stimTS_tier', [(0:K-1).'*hold, seq.']);
  mdl = 'tier_oracle_mdl'; if bdIsLoaded(mdl), close_system(mdl,0); end
  new_system(mdl); load_system(mdl);
  add_block('snn_champions_lib/Donatello_Tier', [mdl '/DUT']);
  set_param([mdl '/DUT'], 'TIER','BALANCED', 'NFRAC','13');   % esplicito: niente default impliciti
  add_block('simulink/Sources/From Workspace', [mdl '/src'], 'VariableName','stimTS_tier', ...
            'SampleTime','1','Interpolate','off','OutputAfterFinalValue','Holding final value');
  add_block('simulink/Signal Routing/Demux', [mdl '/dm'], 'Outputs','4'); add_line(mdl,'src/1','dm/1');
  for j=1:4
    add_block('simulink/Signal Attributes/Data Type Conversion', [mdl '/c' num2str(j)], ...
              'OutDataTypeStr', sprintf('fixdt(1,32,%d)', nfrac));
    add_line(mdl, ['dm/' num2str(j)], ['c' num2str(j) '/1']);
    add_line(mdl, ['c' num2str(j) '/1'], ['DUT/' num2str(j)]);
  end
  add_block('simulink/Signal Routing/Mux', [mdl '/mx'], 'Inputs','5');
  for j=1:5, add_line(mdl, ['DUT/' num2str(j)], ['mx/' num2str(j)]); end
  add_block('simulink/Sinks/To Workspace', [mdl '/Pw'], 'VariableName','Pw','SaveFormat','Array');
  add_line(mdl,'mx/1','Pw/1');
  set_param(mdl,'Solver','FixedStepDiscrete','FixedStep','1','StopTime',num2str(stopT),'SaveOutput','off');
  so = sim(mdl); P = double(so.get('Pw')); close_system(mdl,0);
end
```

function tier_export_vectors(trajList, tag, outdir)
%TIER_EXPORT_VECTORS  Scrive stim/gold .mem + meta per la validazione RTL del Tier@BAL, dal golden FEDELE
%  AL BLOCCO (tier_block_params, il blocco in Simulino). stim = i 4 val (fixdt(1,32,20), 8 hex); gold = i
%  5 param Q7.13 (21b, 6 hex). Valore = STORED INTEGER (compl. a 2) mascherato, via typecast int32->uint32
%  (uint32 di un negativo SATURA a 0 -> falsi mismatch).

  here = fileparts(mfilename('fullpath'));
  mroot = fullfile(here,'..','..','matlab'); addpath(mroot);
  if ~exist(outdir,'dir'), mkdir(outdir); end
  ds = load(fullfile(mroot,'test_dataset.mat')); tr = ds.trajectories;
  Tp = numerictype(1,21,13);
  P = tier_block_params(trajList, HOLD);        % golden dall'oracolo (blocco)
  fs = fopen(fullfile(outdir,['stim_' tag '.mem']),'w');
  fg = fopen(fullfile(outdir,['gold_' tag '.mem']),'w');
  K = 0;
  for i = 1:numel(trajList)
    val = fi(double(tr{trajList(i)}.val), 1, 32, 20);   % la quantizzazione che vede il blocco
    Pi = P{i};
    for k = 1:size(val,2)
      for j = 1:4
        fprintf(fs, '%08X\n', typecast(int32(storedInteger(val(j,k))),'uint32'));
      end
      for c = 1:5
        pf = fi(Pi(k,c), Tp);
        fprintf(fg, '%06X\n', bitand(typecast(int32(storedInteger(pf)),'uint32'), uint32(2^21-1)));
      end
      K = K + 1;
    end
  end
  fclose(fs); fclose(fg);
  save(fullfile(outdir,['meta_' tag '.mat']), 'K', 'trajList');
  fprintf('tier_export_vectors: %d control-step su %d traiettorie -> %s\n', K, numel(trajList), outdir);
end

- [ ] **Step 1: scrivere `tb_tier_stream.v`**

⚠️ **Conferma i nomi porta con l'output della Task 1.** Campiona a fine control-step (`HOLD` clock dopo il
cambio ingresso), misura `LAT` sul 1° control-step. Macro (`KVAL`,`HOLD`,`STIMF`,`GOLDF`) iniettate dal file
`tier_params.vh` (il `-d NAME=val` perde `=` in Git-Bash→`.bat`).

```verilog
`timescale 1ns/1ps
`include "tier_params.vh"
// Pilota Donatello_Tier (VHDL generato) con gli stim, campiona i 5 param a fine control-step vs gold.
module tb_tier_stream;
  localparam integer K    = `KVAL;
  localparam integer HOLD = `HOLD;     // >= latenza RTL (~364); campiona a fine control-step
  reg clk = 0, reset = 0, clk_enable = 1;
  reg  signed [31:0] s, v, dv, v_l;
  wire signed [20:0] v0, T, s0, a, b;  // Q7.13, 21b (confermare nomi/larghezze da Task 1)
  wire ce_out;
  reg  [31:0] stim [0:K*4-1];
  reg  [20:0] gold [0:K*5-1];
  reg  signed [20:0] pw [0:4];
  integer k, i, nmis, lat; reg signed [20:0] prev0;

  Donatello_Tier dut (.clk(clk), .reset(reset), .clk_enable(clk_enable),
     .s(s), .v(v), .dv(dv), .v_l(v_l),
     .v0(v0), .T(T), .s0(s0), .a(a), .b(b), .ce_out(ce_out));

  always #62.5 clk = ~clk;             // 8 MHz (mezzo periodo 62.5 ns)

  initial begin
    $readmemh(`STIMF, stim);
    $readmemh(`GOLDF, gold);
    // 1o control-step presentato PRIMA del de-assert del reset (niente fase a ingresso 0)
    s = stim[0]; v = stim[1]; dv = stim[2]; v_l = stim[3];
    reset = 1; repeat (8) @(posedge clk); reset = 0;
    // misura LAT: primo clock in cui v0 cambia dal valore di reset
    prev0 = v0; lat = 0;
    while (v0 === prev0 && lat < HOLD) begin @(posedge clk); lat = lat + 1; end
    $display("LAT_RTL %0d", lat);
    // completa il 1o control-step fino a fine HOLD, poi campiona e confronta
    repeat (HOLD - lat) @(posedge clk);
    nmis = 0;
    pw[0]=v0; pw[1]=T; pw[2]=s0; pw[3]=a; pw[4]=b;
    for (i=0;i<5;i=i+1) if (pw[i] !== gold[i]) nmis = nmis + 1;
    for (k = 1; k < K; k = k + 1) begin
      s = stim[k*4+0]; v = stim[k*4+1]; dv = stim[k*4+2]; v_l = stim[k*4+3];
      repeat (HOLD) @(posedge clk);
      pw[0]=v0; pw[1]=T; pw[2]=s0; pw[3]=a; pw[4]=b;
      for (i=0;i<5;i=i+1) if (pw[i] !== gold[k*5+i]) nmis = nmis + 1;
    end
    $display("RTLRES nMismatch=%0d n=%0d", nmis, K*5);
    $finish;
  end
endmodule
```

```matlab
function res = run_rtl_validate_tier(mode, trajList)
%RUN_RTL_VALIDATE_TIER  Orchestratore T6-EXACT: genera vettori dal golden-da-blocco, invoca xsim, parse
%  RTLRES + LAT_RTL, assert dei cancelli. mode 'reduced' (subset diversificato, default) o 'full' (1:60).
  here  = fileparts(mfilename('fullpath'));
  mroot = fullfile(here,'..','..','matlab'); addpath(mroot);
  hdir  = fullfile(here,'..','Harness_SNN');
  hdlsrc= fullfile(mroot,'hdlsrc_donatello_tier');
  if nargin<1||isempty(mode), mode='reduced'; end
  if nargin<2||isempty(trajList)
    if strcmp(mode,'full'), trajList = 1:60; else, trajList = subset_diverse(mroot); end
  end
  if ~exist(fullfile(hdlsrc,'Donatello_Tier.vhd'),'file')
    rtl_gen_dut('Donatello_Tier', hdlsrc, 'VHDL', {'TIER','BALANCED','NFRAC','13'});
  end
  tag = mode; HOLD = 500;
  tier_export_vectors(trajList, tag, hdir);
  m = load(fullfile(hdir,['meta_' tag '.mat']));
  cmd = sprintf('bash run_xsim_tier.sh ../../matlab/hdlsrc_donatello_tier %d %d stim_%s.mem gold_%s.mem', ...
                m.K, HOLD, tag, tag);
  [~, out] = system(['cd /d "' hdir '" && ' cmd]);
  tok = regexp(out, 'RTLRES nMismatch=(\d+) n=(\d+)', 'tokens', 'once');
  assert(~isempty(tok), 'xsim non ha prodotto RTLRES. Output:\n%s', out);
  res = struct('nMismatch',str2double(tok{1}), 'n',str2double(tok{2}), 'K',m.K, 'traj',numel(trajList));
  lt = regexp(out,'LAT_RTL (\d+)','tokens','once'); res.lat = str2double(lt{1});
  fprintf('T6-EXACT [%s]: nMismatch = %d / %d  (%d traj x %d control-step x 5 param; lat=%d)\n', ...
          mode, res.nMismatch, res.n, res.traj, m.K, res.lat);
  assert(res.nMismatch==0, ['T6-EXACT FALLITO: il VHDL del Tier NON riproduce il blocco ' ...
         '(%d/%d disallineati) -> non e'' bit-exact a livello RTL.'], res.nMismatch, res.n);
  fprintf('=== T6-EXACT PASSATO [%s]: Donatello_Tier RTL == blocco su %d/%d ===\n', mode, res.n, res.n);
end

function idx = subset_diverse(mroot)
% 1 traj per ogni combinazione scenario x profilo presente + estremi di dv + >=1 con cut-in
  ds = load(fullfile(mroot,'test_dataset.mat')); tr = ds.trajectories; N = numel(tr);
  key = strings(N,1); ci = false(N,1); dvmax = zeros(N,1);
  for t=1:N
    key(t) = string(tr{t}.scenario)+"|"+string(tr{t}.profile);
    c = tr{t}.cut_in; ci(t) = any(double(c(:))~=0);
    v = double(tr{t}.val); dvmax(t) = max(abs(v(3,:)));
  end
  [~,ia] = unique(key,'stable'); idx = ia(:).';            % 1 per combinazione (le 8)
  [~,mx] = max(dvmax); [~,mn] = min(dvmax); idx = unique([idx mx mn]); % estremi dv
  if ~any(ci(idx)), c1 = find(ci,1); idx = unique([idx c1]); end       % almeno un cut-in
  fprintf('subset diversificato: %d traiettorie -> %s\n', numel(idx), mat2str(idx));
end
```
