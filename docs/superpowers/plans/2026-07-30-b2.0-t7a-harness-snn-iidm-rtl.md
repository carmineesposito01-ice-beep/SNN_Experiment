```matlab
function n = t7_export_scenarios(outdir, idx, accelCell)
%T7_EXPORT_SCENARIOS  Scrive, per ogni scenario, i .mem che il TB legge:
%    scen_<i>.mem  : K righe da 1 double (v_leader) in esadecimale a 64 bit ($realtobits)
%    init_<i>.mem  : 4 righe  [s_init; v_init; cut_k (0 = assente); cut_gap]
%    acc_<i>.mem   : K righe di accel (double) -- SOLO per PLANT-PAR: pilota il plant senza DUT
%  accelCell puo' essere [] (allora acc_<i>.mem non viene scritto).
  if ~exist(outdir,'dir'), mkdir(outdir); end
  ml = fileparts(fileparts(fileparts(mfilename('fullpath'))));
  ds = load(fullfile(ml,'matlab','Quantizzation_Study','test_dataset_exhaustive.mat'));
  tr = ds.trajectories;
  n = 0;
  for j = 1:numel(idx)
    t = tr{idx(j)};
    wr(fullfile(outdir,sprintf('scen_%d.mem',j)), double(t.v_leader(:)));
    if isempty(t.cut_in), ck = 0; cg = 0; else, ck = t.cut_in(1); cg = t.cut_in(2); end
    wr(fullfile(outdir,sprintf('init_%d.mem',j)), ...
       [double(t.s_init); double(t.v_init); double(ck); double(cg)]);
    if ~isempty(accelCell)
      wr(fullfile(outdir,sprintf('acc_%d.mem',j)), double(accelCell{j}(:)));
    end
    n = n + 1;
  end
end

function wr(path, x)
% double -> 16 cifre esadecimali (bit pattern IEEE-754), leggibile da $readmemh + $bitstoreal
  f = fopen(path,'w');
  for i = 1:numel(x), fprintf(f, '%016s\n', lower(dec2hex(typecast(double(x(i)),'uint64'),16))); end
  fclose(f);
end
```

- [ ] **Step 3: scrivere il TB del solo plant**

`FaseB2.0/Harness_SNN_IIDM/tb_plant_only.v` — riproduce `qz_cl_sim` riga per riga, **senza DUT**:
```verilog
`timescale 1ns/1ps
`include "t7_params.vh"          // `KVAL, `SCENF, `INITF, `ACCF, `OUTF
// PLANT-PAR (T7a Task 1): il plant del TB riproduce qz_cl_sim? Pilotato con la sequenza accel del
// riferimento, SENZA DUT: isola i difetti d'integrazione prima dell'anello live.
// Ordine ESATTO di qz_cl_sim: cut-in -> vl -> dv (v CORRENTE) -> registra PRE-update -> v -> s (con la
// NUOVA v) -> break se s<=0.
module tb_plant_only;
  localparam integer K = `KVAL;
  localparam real DT = 0.1;
  reg [63:0] vlm[0:K-1], initm[0:3], accm[0:K-1];
  real s, v, vl, dv, acc, vnew, cut_gap;
  integer k, cut_k, last, collided, fo;
  real impact_dv;
  initial begin
    $readmemh(`SCENF, vlm); $readmemh(`INITF, initm); $readmemh(`ACCF, accm);
    s = $bitstoreal(initm[0]); v = $bitstoreal(initm[1]);
    cut_k = $rtoi($bitstoreal(initm[2])); cut_gap = $bitstoreal(initm[3]);
    collided = 0; impact_dv = 0.0; last = K;
    fo = $fopen(`OUTF, "w");
    for (k = 1; k <= K; k = k + 1) begin
      if (cut_k != 0 && k == cut_k) s = cut_gap;      // teletrasporto (k e' 1-based, come MATLAB)
      vl  = $bitstoreal(vlm[k-1]);
      dv  = v - vl;                                   // v CORRENTE, prima dell'update
      acc = $bitstoreal(accm[k-1]);                   // niente DUT: accel dal riferimento
      $fwrite(fo, "%0d %h %h %h %h\n", k, $realtobits(s), $realtobits(v),
              $realtobits(vl), $realtobits(dv));      // registra PRE-update
      vnew = v + acc*DT; if (vnew < 0.0) vnew = 0.0;
      s = s + (vl - vnew)*DT;                         // usa la NUOVA v
      v = vnew;
      if (s <= 0.0) begin
        collided = 1; impact_dv = (v - vl) > 0.0 ? (v - vl) : 0.0; last = k;
        k = K + 1;                                    // esce dal for: equivale al break di MATLAB
      end
    end
    $fwrite(fo, "END %0d %0d %h %h\n", last, collided, $realtobits(s), $realtobits(impact_dv));
    $fclose(fo);
    $display("PLANTPAR last=%0d collided=%0d", last, collided);
    $finish;
  end
endmodule
```

`FaseB2.0/Harness_SNN_IIDM/t7_plant_par.m`:
```matlab
function n = t7_plant_par(workdir, O, idx)
%T7_PLANT_PAR  Cancello PLANT-PAR sull'intero perimetro: il plant del TB riproduce qz_cl_sim
%  bit-per-bit, su ogni scenario, pilotato con la sequenza accel del riferimento e SENZA DUT.
%  Verilog `real` e' IEEE-754 double come MATLAB: la parita' dev'essere ESATTA, non "al livello del float".
  n = 0;
  for i = 1:numel(idx)
    o = O{i};
    T = readlines(fullfile(workdir, sprintf('plant_%d.txt', i)));
    T = T(strlength(T) > 0);
    body = T(~startsWith(T,"END"));
    h2d = @(h) t7_hexread(h);   % NON hex2dec: perde precisione oltre 2^53 (vedi t7_hexread.m)
    m = numel(body);
    S = zeros(m,1); V = zeros(m,1); VL = zeros(m,1); DV = zeros(m,1);
    for r = 1:m
      f = split(strtrim(body(r)));
      S(r) = h2d(f(2)); V(r) = h2d(f(3)); VL(r) = h2d(f(4)); DV(r) = h2d(f(5));
    end
    k = min(m, o.N);
    d = sum([S(1:k) ~= o.series.s(1:k).'; V(1:k) ~= o.series.v(1:k).'; ...
             VL(1:k) ~= o.series.vl(1:k).'; DV(1:k) ~= o.series.dv(1:k).']);
    if m ~= o.N, d = d + 1; end          % anche la lunghezza (troncamento su collisione) deve coincidere
    n = n + d;
  end
  fprintf('PLANT-PAR    : %d disallineamenti su %d scenari\n', n, numel(idx));
end
```

```verilog
`timescale 1ns/1ps
`include "t7_params.vh"          // `KVAL, `HOLDV, `SCENF, `INITF, `OUTF
// T7a: anello chiuso LIVE. Plant (provato da PLANT-PAR) <-> Donatello_SNN_IIDM in retroazione sull'accel.
// Ogni control-step: stato plant -> quantizza a sfix32_En20 (FLOOR) -> pilota il DUT -> attende HOLD ->
// legge accel (sfix13_En8) e i 5 params (segnali del top) -> il plant integra.
// Le SERIE esportate sono la sorgente dei numeri riportati: stato FISICO non quantizzato + uscite del DUT.
module tb_snn_iidm_closed;
  localparam integer K = `KVAL;
  localparam integer HOLD = `HOLDV;
  localparam real DT = 0.1;
  reg [63:0] vlm[0:K-1], initm[0:3];
  real s, v, vl, dv, acc, vnew, cut_gap, impact_dv;
  integer k, cut_k, last, collided, fo;
  reg clk = 0, reset = 1, ce = 1;
  reg [31:0] s_in, v_in, dv_in, vl_in;
  wire [12:0] accel;
  wire ce_out;

  Donatello_SNN_IIDM dut(.clk(clk), .reset(reset), .clk_enable(ce),
                         .s(s_in), .v(v_in), .dv(dv_in), .v_l(vl_in),
                         .ce_out(ce_out), .accel(accel));
  always #5 clk = ~clk;

  // FLOOR a 2^-20, poi stored-integer a 32 bit: la convenzione fissata in spec §4.
  function [31:0] toq(input real x);
    begin toq = $rtoi($floor(x * 1048576.0)); end
  endfunction
  function real fromq(input signed [31:0] q); begin fromq = $itor(q) / 1048576.0; end endfunction

  task drive;                                        // stato plant -> ingressi del DUT
    begin s_in = toq(s); v_in = toq(v); dv_in = toq(dv); vl_in = toq(vl); end
  endtask

  initial begin
    $readmemh(`SCENF, vlm); $readmemh(`INITF, initm);
    s = $bitstoreal(initm[0]); v = $bitstoreal(initm[1]);
    cut_k = $rtoi($bitstoreal(initm[2])); cut_gap = $bitstoreal(initm[3]);
    collided = 0; impact_dv = 0.0; last = K;
    fo = $fopen(`OUTF, "w");
    // primo control-step presentato PRIMA del rilascio del reset: il DUT non deve vedere
    // una transizione da indefinito (inferenza spuria — trappola T6b).
    k = 1;
    if (cut_k == 1) s = cut_gap;
    vl = $bitstoreal(vlm[0]); dv = v - vl; drive;
    repeat (8) @(posedge clk); reset = 0;
    for (k = 1; k <= K; k = k + 1) begin
      if (k > 1) begin
        if (cut_k != 0 && k == cut_k) s = cut_gap;
        vl = $bitstoreal(vlm[k-1]);
        dv = v - vl;
        drive;
      end
      repeat (HOLD) @(posedge clk);
      acc = $itor($signed(accel)) / 256.0;           // sfix13_En8 -> real
      // serie: stato FISICO (per le metriche) + uscite del DUT (accel e 5 params)
      $fwrite(fo, "%0d %h %h %h %h %h %0d %0d %0d %0d %0d\n", k,
              $realtobits(s), $realtobits(v), $realtobits(vl), $realtobits(dv), $realtobits(acc),
`ifdef NETLIST_SIM
              0, 0, 0, 0, 0);                        // la netlist perde i nomi gerarchici (HDL_PHASE §9)
`else
              $signed(dut.Tier_v0), $signed(dut.Tier_T), $signed(dut.Tier_s0),
              $signed(dut.Tier_a),  $signed(dut.Tier_b));
`endif
      vnew = v + acc*DT; if (vnew < 0.0) vnew = 0.0;
      s = s + (vl - vnew)*DT;
      v = vnew;
      if (s <= 0.0) begin
        collided = 1; impact_dv = (v - vl) > 0.0 ? (v - vl) : 0.0; last = k;
        k = K + 1;
      end
    end
    $fwrite(fo, "END %0d %0d %h %h\n", last, collided, $realtobits(s), $realtobits(impact_dv));
    $fclose(fo);
    $display("T7RES last=%0d collided=%0d", last, collided);
    $finish;
  end
endmodule
```

```matlab
function R = t7_read_series(path)
%T7_READ_SERIES  Legge ser_<i>.txt prodotto da tb_snn_iidm_closed.v.
%  Righe: k s v vl dv accel  p_v0 p_T p_s0 p_a p_b   (i primi 5 in esadecimale a 64 bit; i params
%  come stored integer sfix21_En13). Ultima riga: END last collided s_finale impact_dv.
  txt = readlines(path);
  txt = txt(strlength(txt) > 0);
  isEnd = startsWith(txt, "END");
  body = txt(~isEnd); tail = txt(isEnd);
  assert(numel(tail) == 1, '%s: attesa UNA riga END, trovate %d', path, numel(tail));
  n = numel(body);
  R = struct('k',zeros(n,1),'s',zeros(n,1),'v',zeros(n,1),'vl',zeros(n,1), ...
             'dv',zeros(n,1),'a',zeros(n,1),'params',zeros(n,5));
  h2d = @(h) t7_hexread(h);   % NON hex2dec: perde precisione oltre 2^53 (vedi t7_hexread.m)
  for i = 1:n
    f = split(strtrim(body(i)));
    assert(numel(f) == 11, '%s riga %d: attesi 11 campi, trovati %d', path, i, numel(f));
    R.k(i)  = double(f(1));
    R.s(i)  = h2d(f(2)); R.v(i) = h2d(f(3)); R.vl(i) = h2d(f(4));
    R.dv(i) = h2d(f(5)); R.a(i) = h2d(f(6));
    R.params(i,:) = double(f(7:11)).' / 8192.0;          % sfix21_En13
  end
  g = split(strtrim(tail(1)));
  R.N = double(g(2)); R.collided = logical(double(g(3)));
  R.s_end = h2d(g(4)); R.impact_dv = h2d(g(5));
  assert(R.N == n, '%s: END dichiara N=%d ma le righe sono %d', path, R.N, n);
end
```

```matlab
function V = t7_rtl_validate(workdir, G, idx)

%   T7-EXACT     : accel RTL == accel del golden, bit-esatto, su tutti gli scenari
%   PLANT-EXACT  : s,v,vl,dv del TB == quelle del golden (il plant si comporta uguale ANCHE in anello)
%   PARAM-RANGE  : i 5 params stanno nei limiti del decode [8 .5 1 .3 .5]..[45 2.5 5 2.5 3]
%   LEN-EXACT    : la lunghezza dello scenario (troncamento su collisione) coincide
  LO = [8 0.5 1 0.3 0.5]; HI = [45 2.5 5 2.5 3];       % limiti del decode (acciidm_m_algo, commento riga 649)
  nExact = 0; nTot = 0; nPlant = 0; nPlantTot = 0; nRange = 0; nLen = 0; nRep = 0; repScen = [];
  first = struct('scen',[],'k',[],'rtl',[],'gold',[]);
  for i = 1:numel(idx)
    R = t7_read_series(fullfile(workdir, sprintf('ser_%d.txt', i)));
    g = G{i};
    if R.N ~= g.N, nLen = nLen + 1; end
    n = min(R.N, g.N);
    dA = R.a(1:n) - g.series.a(1:n).';
    bad = find(dA ~= 0);
    nExact = nExact + numel(bad); nTot = nTot + n;
    if ~isempty(bad) && isempty(first.scen)
      first.scen = idx(i); first.k = bad(1);
      first.rtl = R.a(bad(1)); first.gold = g.series.a(bad(1));
    end
    dP = [R.s(1:n)-g.series.s(1:n).', R.v(1:n)-g.series.v(1:n).', ...
          R.vl(1:n)-g.series.vl(1:n).', R.dv(1:n)-g.series.dv(1:n).'];
    nPlant = nPlant + sum(dP(:) ~= 0); nPlantTot = nPlantTot + numel(dP);
    P = R.params(1:n,:);
    nRange = nRange + sum(any(P < LO | P > HI, 2));

    nRep = nRep + rep;
    if rep > 0, repScen(end+1) = idx(i); end %#ok<AGROW>
  end
  V = struct('nExact',nExact,'nTot',nTot,'nPlant',nPlant,'nPlantTot',nPlantTot, ...

  fprintf('LEN-EXACT    : %d scenari con lunghezza diversa dal golden\n', nLen);
  fprintf('NO-REPEAT    : %d control-step con i 5 parametri ripetuti', nRep);
```

```matlab
function out = t7_series_to_mat(workdir, O, idx, outfile)
%T7_SERIES_TO_MAT  Impacchetta per Python: serie RTL (sorgente dei numeri riportati) + serie ORACOLO.
%  Le 9 chiavi richieste da all_metrics sono scritte TUTTE ed ESPLICITAMENTE (spec §5): omettere
%  impact_dv non darebbe errore, darebbe 0 — un numero credibile e sbagliato.
  RTL = struct('name',{},'s',{},'v',{},'vl',{},'dv',{},'a_ego',{},'params',{}, ...
               'collided',{},'min_gap',{},'impact_dv',{});
  ORA = RTL;
  for i = 1:numel(idx)
    R = t7_read_series(fullfile(workdir, sprintf('ser_%d.txt', i)));
    n = R.N;
    if R.collided, mg = R.s_end; else, mg = min(R.s(1:n)); end   % come qz_cl_sim riga 33
    RTL(i) = pack(sprintf('scen_%d', idx(i)), R.s(1:n), R.v(1:n), R.vl(1:n), R.dv(1:n), ...
                  R.a(1:n), R.params(1:n,:), R.collided, mg, R.impact_dv);
    o = O{i};
    ORA(i) = pack(sprintf('scen_%d', idx(i)), o.series.s, o.series.v, o.series.vl, o.series.dv, ...
                  o.series.a, o.params, o.collided, o.min_gap, o.impact_dv);
  end
  save(outfile, 'RTL', 'ORA', '-v7');
  out = outfile;
  fprintf('t7_series_to_mat: %d scenari -> %s\n', numel(idx), outfile);
end

function e = pack(name, s, v, vl, dv, a, p, coll, mg, idv)
```

`FaseB2.0/Harness_SNN_IIDM/t7_metrics.py`:
```python
#!/usr/bin/env python3
"""T7a — metriche di car-following dal motore CANONICO, sulle serie prodotte dall'RTL.

Non riscrive alcuna metrica: usa utils.closed_loop_eval, lo stesso motore che ha prodotto
VALIDATION_REPORT_v3 e QUANTIZATION_STUDY_REPORT -> numeri confrontabili per costruzione.

CONTRATTO: all_metrics vuole NOVE chiavi. Una di esse, impact_dv, e' letta con
traj.get('impact_dv', 0.0): ometterla NON da' errore, restituisce 0.0 -- cioe' "collisione a
severita' nulla" per una collisione reale. Qui le nove chiavi sono ASSERITE prima della chiamata.

Uso:  python t7_metrics.py <series.mat> <out.json>
"""
import json
import sys

import numpy as np
import scipy.io as sio

sys.path.insert(0, '.')
from utils.closed_loop_eval import all_metrics, string_stability_gain

REQUIRED = ('s', 'v', 'vl', 'dv', 'a_ego', 'params', 'collided', 'min_gap', 'impact_dv')


def to_traj(e):
    """Costruisce il dict richiesto dal motore, asserendo le nove chiavi."""
    t = {
        's': np.atleast_1d(e.s).astype(float),
        'v': np.atleast_1d(e.v).astype(float),
        'vl': np.atleast_1d(e.vl).astype(float),
        'dv': np.atleast_1d(e.dv).astype(float),
        'a_ego': np.atleast_1d(e.a_ego).astype(float),
        'params': np.atleast_2d(e.params).astype(float),
        'collided': bool(e.collided),
        'min_gap': float(e.min_gap),
        'impact_dv': float(e.impact_dv),
    }
    missing = [k for k in REQUIRED if k not in t]
    if missing:
        raise AssertionError(f'contratto violato: chiavi mancanti {missing}')
    n = len(t['s'])
    for k in ('v', 'vl', 'dv', 'a_ego'):
        if len(t[k]) != n:
            raise AssertionError(f'{k}: lunghezza {len(t[k])} != {n}')
    if t['params'].shape != (n, 5):
        raise AssertionError(f"params: forma {t['params'].shape} != {(n, 5)}")
    return t


def metrics_of(e):
    t = to_traj(e)
    m = dict(all_metrics(t))
    m['string_stability'] = string_stability_gain(t)
    return m


def main(matfile, outfile):
    d = sio.loadmat(matfile, squeeze_me=True, struct_as_record=False)
    res = {}
    for tag in ('RTL', 'ORA'):
        arr = np.atleast_1d(d[tag])
        res[tag] = {str(e.name): metrics_of(e) for e in arr}
    # collisioni AGGIUNTIVE rispetto all'oracolo: il cancello DURO di T7 (spec §7)
    extra = [k for k in res['RTL']
             if res['RTL'][k]['collided'] and not res['ORA'][k]['collided']]
    res['_summary'] = {
        'n_scenari': len(res['RTL']),
        'coll_rtl': sum(1 for k in res['RTL'] if res['RTL'][k]['collided']),
        'coll_oracolo': sum(1 for k in res['ORA'] if res['ORA'][k]['collided']),
        'coll_extra': len(extra),
        'coll_extra_scenari': extra,
        'n_metriche': len(next(iter(res['RTL'].values()))),
    }
    with open(outfile, 'w', encoding='utf-8') as f:
        json.dump(res, f, indent=2, sort_keys=True, default=float)
    s = res['_summary']
    print(f"scenari={s['n_scenari']}  metriche/scenario={s['n_metriche']}")
    print(f"collisioni RTL={s['coll_rtl']}  oracolo={s['coll_oracolo']}  EXTRA={s['coll_extra']}")
    if extra:
        print(f"  scenari con collisione extra: {extra}")
    return 0


if __name__ == '__main__':
    sys.exit(main(sys.argv[1], sys.argv[2]))
```

```matlab
function sensitivity_t7(workdir, idx)

  Gp = G; Gp{1}.N = Gp{1}.N - 1;
  V = t7_rtl_validate(workdir, Gp, idx(1));

  fprintf('LEN-EXACT    sensibile: rilevata la lunghezza diversa\n');

  % PARAM-RANGE: un parametro fuori dominio nella serie letta
  R = t7_read_series(fullfile(workdir, sprintf('ser_%d.txt', 1)));

  assert(sum(any(P < LO | P > HI, 2)) == 1, 'PARAM-RANGE NON sensibile');
  fprintf('PARAM-RANGE  sensibile: rilevato il parametro fuori dominio\n');

  % NO-REPEAT: duplicando una riga di parametri il conteggio DEVE salire di 1;

  P = R.params; P(10,:) = P(9,:);
```

   **falsa** su questo blocco: non copiarla.
6. **Macro al TB via file `.vh`**, mai `-d NOME=val`: il wrapper Git-Bash→`.bat` mangia l'`=`.
